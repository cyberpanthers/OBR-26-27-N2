#include <Servo.h>
#include <SoftwareSerial.h>

SoftwareSerial BT(2, 3); // RX, TX

int ldr = A0;
int sensibilidade = 540;
int val = 0;

Servo motor;

int STOP = 87;
int GIRA = 0;

const int trigPin = 8;
const int echoPin = 9;

// Bluetooth
const int statePin = 5; // pino STATE do HC-05
const int ledPin = 4;   // LED indicador

bool sinalPendente = false;

float distancia() {

  digitalWrite(trigPin, LOW);
  delayMicroseconds(2);

  digitalWrite(trigPin, HIGH);
  delayMicroseconds(10);

  digitalWrite(trigPin, LOW);

  long duracao = pulseIn(echoPin, HIGH);

  return duracao * 0.034 / 2;
}

void setup() {

  Serial.begin(9600);
  BT.begin(9600);

  pinMode(ldr, INPUT);

  pinMode(trigPin, OUTPUT);
  pinMode(echoPin, INPUT);

  pinMode(statePin, INPUT);
  pinMode(ledPin, OUTPUT);

  motor.attach(6);
  motor.write(STOP);

  Serial.println("Sistema iniciado");
}

void loop() {

  // Acende LED quando Bluetooth conectado
  digitalWrite(ledPin, digitalRead(statePin));

  // Se houver mensagem pendente e Bluetooth conectar
  if (sinalPendente && digitalRead(statePin) == HIGH) {
    BT.println("B");
    Serial.println("Bluetooth conectado - sinal B enviado");
    sinalPendente = false;
  }

  val = analogRead(ldr);
  Serial.println(val);

  if (val > sensibilidade) {

    Serial.println("Portal 1 aberto");

    motor.write(GIRA);
    delay(1000);
    motor.write(STOP);

    Serial.println("Aguardando cubo...");

    while (distancia() > 5) {
      delay(10);
    }

    Serial.println("Cubo detectado");

    // Se conectado, envia imediatamente
    if (digitalRead(statePin) == HIGH) {
      BT.println("B");
      Serial.println("Sinal B enviado");
    }
    // Se desconectado, guarda para enviar depois
    else {
      sinalPendente = true;
      Serial.println("Bluetooth desconectado - sinal armazenado");
    }

    // Espera o cubo sair
    while (analogRead(ldr) > sensibilidade) {
      delay(100);
    }
  }

  delay(100);
}
