#include <SoftwareSerial.h>

SoftwareSerial BT(10,11);

#define led1 12
#define led2 13
#define led3 9
#define led4 8

#define buzzer 7

void setup()
{
  BT.begin(9600);

  pinMode(led1, OUTPUT);
  pinMode(led2, OUTPUT);
  pinMode(led3, OUTPUT);
  pinMode(led4, OUTPUT);

  pinMode(buzzer, OUTPUT);
}

void loop()
{
  if(BT.available())
  {
      char c = BT.read();

      if(c == 'A')
      {
        for(int i = 0; i < 10; i++)
        {
          tone(buzzer, 500);

          digitalWrite(led1, HIGH);
          digitalWrite(led3, HIGH);
          digitalWrite(led2, LOW);
          digitalWrite(led4, LOW);
          delay(500);

          noTone(buzzer);
         
          digitalWrite(led1, LOW);
          digitalWrite(led3, LOW);
          digitalWrite(led2, HIGH);
          digitalWrite(led4, HIGH);
          delay(500);
       
        }

        noTone(buzzer);
        digitalWrite(led1, LOW);
        digitalWrite(led2, LOW);
        digitalWrite(led3, LOW);
        digitalWrite(led4, LOW);
       
    }
    else
    {
      noTone(buzzer);

        digitalWrite(led1, LOW);
        digitalWrite(led2, LOW);
        digitalWrite(led3, LOW);
        digitalWrite(led4, LOW);
    }
  }
}
