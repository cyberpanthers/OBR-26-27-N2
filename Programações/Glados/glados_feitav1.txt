import mediapipe as mp

import time

import subprocess


PASTA_AUDIOS = "falas_atualizadas"


def aquecer_audio():
    print("Aquecendo JBL...")

    processo = subprocess.Popen(
        ["pw-play", os.path.join(PASTA_AUDIOS, "silencio.wav")],
        stdout=subprocess.DEVNULL,
        stderr=subprocess.DEVNULL
    )

    processo.wait()

    time.sleep(1)

    print("JBL pronta!")


def tocar_audio(nome):
    caminho = os.path.join(PASTA_AUDIOS, nome)

    print(f"Tocando {nome}")

    processo = subprocess.Popen(
        ["pw-play", caminho],
        stdout=subprocess.DEVNULL,
        stderr=subprocess.DEVNULL
    )

    processo.wait()

try:

    ser = serial.Serial('/dev/ttyUSB0', 115200 , timeout=1)

    time.sleep(2)

    print("Arduino conectado com sucesso!")

except Exception as e:

    print(f"Rodando sem Arduino. Erro: {e}")

    ser = None



def enviar_comando(comando):
    if not ser:
        print(comando)
        return

    print(">>", comando)

    #ser.reset_input_buffer()
    ser.write((comando + "\n").encode())
    ser.flush()

    while True:
        resposta = ser.readline().decode(errors="ignore").strip()

        if resposta:
            print("Recebido:", repr(resposta))

        if resposta == "Concluído.":
            break

# MEDIAPIPE

# ==========================



mp_hands = mp.solutions.hands

mp_draw = mp.solutions.drawing_utils



cap = cv2.VideoCapture(0)



# ==========================

# INICIA O SERVIDOR FLASK

# ==========================



threading.Thread(

    target=flask_app.app.run,

    kwargs={

        "host": "0.0.0.0",

        "port": 5000,

        "debug": False,

        "use_reloader": False

    },

    daemon=True

).start()





print("Sistema iniciado!")


with mp_hands.Hands(
    min_detection_confidence=0.6,
    min_tracking_confidence=0.6
) as hands:
        
        while cap.isOpened():

            ret, frame = cap.read()

            if not ret:
                print("Erro ao capturar imagem.")
                break

            frame = cv2.flip(frame, 1)
            rgb = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)

            results = hands.process(rgb)

            if results.multi_hand_landmarks:

                for hand_landmarks in results.multi_hand_landmarks:

                    mp_draw.draw_landmarks(
                        frame,
                        hand_landmarks,
                        mp_hands.HAND_CONNECTIONS
                    )

                    ponta_indicador = hand_landmarks.landmark[8].y
                    ponta_medio = hand_landmarks.landmark[12].y
                    ponta_anelar = hand_landmarks.landmark[16].y
                    ponta_minimo = hand_landmarks.landmark[20].y

                    base_indicador = hand_landmarks.landmark[6].y
                    base_medio = hand_landmarks.landmark[10].y
                    base_anelar = hand_landmarks.landmark[14].y
                    base_minimo = hand_landmarks.landmark[18].y

                    if (
                        ponta_indicador < base_indicador and
                        ponta_medio < base_medio and
                        ponta_anelar < base_anelar and
                        ponta_minimo < base_minimo
                    ):

                        print("Mão aberta detectada!")

                        cap.release()
                        cv2.destroyAllWindows()

                        executar_apresentacao()

                        if ser:
                            ser.close()

                        exit()

            cv2.putText(
                frame,
                "AGUARDANDO MAO ABERTA",
                (20, 40),
                cv2.FONT_HERSHEY_SIMPLEX,
                0.8,
                (0, 255, 0),
                2
            )

            flask_app.frame_global = frame.copy()

            time.sleep(0.001)


# FINALIZAÇÃO

# ==========================

cap.release()

if ser:
    ser.close()

cv2.destroyAllWindows()
