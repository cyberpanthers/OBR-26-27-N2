import cv2
import threading
import flask_app
import mediapipe as mp
import serial
import time
import random
import os
import pygame

# ==========================
# CONFIGURAÇÃO DOS ÁUDIOS
# ==========================

pygame.mixer.init()

PASTA_CORINGAS = "Voice core/Coringas"  

audios_coringa = [
    os.path.join(PASTA_CORINGAS, arq)
    for arq in os.listdir(PASTA_CORINGAS)
    if arq.lower().endswith(".wav")
]

ultimo_audio = None


def tocar_audio_aleatorio():
    global ultimo_audio

    if len(audios_coringa) == 0:
        print("Nenhum áudio encontrado!")
        return

    disponiveis = [a for a in audios_coringa if a != ultimo_audio]

    if len(disponiveis) == 0:
        disponiveis = audios_coringa

    audio = random.choice(disponiveis)
    ultimo_audio = audio

    print(f"Tocando: {os.path.basename(audio)}")

    pygame.mixer.music.load(audio)
    pygame.mixer.music.play()


# ==========================
# CONFIGURAÇÃO DO ARDUINO
# ==========================

try:
    ser = serial.Serial('/dev/ttyUSB0', 115200 , timeout=1)
    time.sleep(2)
    print("Arduino conectado com sucesso!")
except Exception as e:
    print(f"Rodando sem Arduino. Erro: {e}")
    ser = None


# ==========================
# MEDIAPIPE
# ==========================

mp_hands = mp.solutions.hands
mp_draw = mp.solutions.drawing_utils

cap = cv2.VideoCapture(0)

# ==========================
# INICIA O SERVIDOR FLASK
# ==========================

threading.Thread(
    target=flask_app.app.run,
    kwargs={
        "host": "0.0.0.0",
        "port": 5000,
        "debug": False,
        "use_reloader": False
    },
    daemon=True
).start()

tempo_bloqueio = 0
etapa = 1

print("Sistema iniciado!")

with mp_hands.Hands(
        min_detection_confidence=0.6,
        min_tracking_confidence=0.6) as hands:

    while cap.isOpened():

        ret, frame = cap.read()

        if not ret:
            print("Erro ao capturar imagem.")
            break

        frame = cv2.flip(frame, 1)
        rgb = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)

        tempo_atual = time.time()

        # ==========================
        # SISTEMA CONGELADO
        # ==========================

        if tempo_atual < tempo_bloqueio:

            restante = int(tempo_bloqueio - tempo_atual)

            status = f"CONGELADO: {restante}s"

            comando_ativo = True

        else:

            status = "AGUARDANDO GESTO"

            comando_ativo = False

            results = hands.process(rgb)

            if results.multi_hand_landmarks:

                for hand_landmarks in results.multi_hand_landmarks:

                    mp_draw.draw_landmarks(
                        frame,
                        hand_landmarks,
                        mp_hands.HAND_CONNECTIONS
                    )

                    # Pontas
                    ponta_indicador = hand_landmarks.landmark[8].y
                    ponta_medio = hand_landmarks.landmark[12].y
                    ponta_anelar = hand_landmarks.landmark[16].y
                    ponta_minimo = hand_landmarks.landmark[20].y

                    # Bases
                    base_indicador = hand_landmarks.landmark[6].y
                    base_medio = hand_landmarks.landmark[10].y
                    base_anelar = hand_landmarks.landmark[14].y
                    base_minimo = hand_landmarks.landmark[18].y

                    # ==========================
                    # MÃO ABERTA
                    # ==========================

                    if (
                        ponta_indicador < base_indicador and
                        ponta_medio < base_medio and
                        ponta_anelar < base_anelar and
                        ponta_minimo < base_minimo
                    ):

                        print("Mão aberta detectada!")

                        # Toca um áudio aleatório
                        tocar_audio_aleatorio()

                        # Envia comando ao Arduino
                        if ser:
                            try:
                                ser.write(f"{etapa}\n".encode())
                                ser.flush()

                                print(f"Etapa {etapa} enviada.")

                                etapa += 1

                                if etapa > 4:
                                    etapa = 1

                            except Exception as e:
                                print(e)

                        # Bloqueia por 10 segundos
                        tempo_bloqueio = time.time() + 10

                        break

        # ==========================
        # INTERFACE
        # ==========================

        cor = (0, 0, 255) if comando_ativo else (0, 255, 0)

        cv2.putText(
            frame,
            status,
            (20, 40),
            cv2.FONT_HERSHEY_SIMPLEX,
            0.8,
            cor,
            2
        )


        
        # envia o frame para o Flask
        flask_app.frame_global = frame.copy()

        time.sleep(0.001)

# MUITO IMPORTANTE
flask_app.frame_global = frame.copy()

time.sleep(0.001)
# FINALIZAÇÃO
# ==========================

cap.release()

if ser:
    ser.close()

pygame.mixer.quit()

cv2.destroyAllWindows()
